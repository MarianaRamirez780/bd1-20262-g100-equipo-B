CREATE TABLE rol (
    id_rol SMALLSERIAL NOT NULL,
    nombre VARCHAR(30) NOT NULL,
    descripcion VARCHAR(200),
    CONSTRAINT pk_rol PRIMARY KEY (id_rol)
);

CREATE TABLE tipo_usuario (
    id_tipo_usuario SMALLSERIAL NOT NULL,
    nombre VARCHAR(30) NOT NULL,
    descripcion VARCHAR(200),
    CONSTRAINT pk_tipo_usuario PRIMARY KEY (id_tipo_usuario)
);

CREATE TABLE dependencia (
    id_depa SMALLSERIAL NOT NULL,
    nombre VARCHAR(50) NOT NULL,
    CONSTRAINT pk_dependencia PRIMARY KEY (id_depa)
);

CREATE TABLE usuario (
    id_usuario SERIAL NOT NULL,
    id_rol SMALLINT NOT NULL,
    nombre_1 VARCHAR(30) NOT NULL,
    nombre_2 VARCHAR(30),
    apellido_1 VARCHAR(30) NOT NULL,
    apellido_2 VARCHAR(30) NOT NULL,
    fecha_nac DATE NOT NULL,
    edad SMALLINT NOT NULL,
    sexo CHAR(1),
    correo VARCHAR(50) NOT NULL,
    contrasena VARCHAR(25) NOT NULL,
    telefono VARCHAR(15),
    fecha_registro TIMESTAMP NOT NULL,
    CONSTRAINT pk_usuario PRIMARY KEY (id_usuario),
    CONSTRAINT fk_usuario_rol FOREIGN KEY (id_rol) REFERENCES rol(id_rol),
    CONSTRAINT uk_usuario_correo UNIQUE (correo),
    CONSTRAINT chk_usuario_edad CHECK (edad >= 0 AND edad <= 150)
);

CREATE TABLE usuario_tipo_usuario (
    id_usuario INTEGER NOT NULL,
    id_tipo_usuario SMALLINT NOT NULL,
    CONSTRAINT pk_usuario_tipo_usuario PRIMARY KEY (id_usuario, id_tipo_usuario),
    CONSTRAINT fk_usu_tipo_usuario FOREIGN KEY (id_usuario) REFERENCES usuario(id_usuario),
    CONSTRAINT fk_usu_tipo_tipousu FOREIGN KEY (id_tipo_usuario) REFERENCES tipo_usuario(id_tipo_usuario)
);

CREATE TABLE usuario_dependencia (
    id_depa SMALLINT NOT NULL,
    id_usuario INTEGER NOT NULL,
    CONSTRAINT pk_usuario_dependencia PRIMARY KEY (id_depa, id_usuario),
    CONSTRAINT fk_usu_depa_dependencia FOREIGN KEY (id_depa) REFERENCES dependencia(id_depa),
    CONSTRAINT fk_usu_depa_usuario FOREIGN KEY (id_usuario) REFERENCES usuario(id_usuario)
);

CREATE TABLE grupo (
    id_grupo SERIAL NOT NULL,
    nombre VARCHAR(50) NOT NULL,
    id_perfil INTEGER NOT NULL,
    descipcion VARCHAR(500),
    fecha_creacion TIMESTAMP NOT NULL,
    CONSTRAINT pk_grupo PRIMARY KEY (id_grupo)
);

CREATE TABLE perfil (
    id_perfil SERIAL NOT NULL,
    id_usuario INTEGER NOT NULL,
    id_grupo INTEGER NOT NULL,
    biografia VARCHAR(500),
    foto CHARACTER VARYING(1000) NOT NULL,
    perfil_extra VARCHAR(4000),
    fecha_perfil TIMESTAMP NOT NULL,
    CONSTRAINT pk_perfil PRIMARY KEY (id_perfil),
    CONSTRAINT fk_perfil_usuario FOREIGN KEY (id_usuario) REFERENCES usuario(id_usuario),
    CONSTRAINT fk_perfil_grupo FOREIGN KEY (id_grupo) REFERENCES grupo(id_grupo)
);

CREATE TABLE usuario_grupo (
    id_usuario INTEGER NOT NULL,
    id_grupo INTEGER NOT NULL,
    CONSTRAINT pk_usuario_grupo PRIMARY KEY (id_usuario, id_grupo),
    CONSTRAINT fk_usu_grupo_usuario FOREIGN KEY (id_usuario) REFERENCES usuario(id_usuario),
    CONSTRAINT fk_usu_grupo_grupo FOREIGN KEY (id_grupo) REFERENCES grupo(id_grupo)
);

CREATE TABLE publicacion (
    id_publicacion BIGSERIAL NOT NULL,
    id_usuario INTEGER NOT NULL,
    id_perfil INTEGER NOT NULL,
    titulo VARCHAR(50) NOT NULL,
    texto VARCHAR(5000),
    URL VARCHAR(1000),
    imagen CHARACTER VARYING(1000),
    fecha_publicacion TIMESTAMP NOT NULL,
    CONSTRAINT pk_publicacion PRIMARY KEY (id_publicacion),
    CONSTRAINT fk_publicacion_usuario FOREIGN KEY (id_usuario) REFERENCES usuario(id_usuario),
    CONSTRAINT fk_publicacion_perfil FOREIGN KEY (id_perfil) REFERENCES perfil(id_perfil)
);

CREATE TABLE comentario (
    id_comentario BIGSERIAL NOT NULL,
    id_usuario INTEGER NOT NULL,
    id_publicacion BIGINT NOT NULL,
    texto VARCHAR(2000),
    fecha_comentario TIMESTAMP NOT NULL,
    CONSTRAINT pk_comentario PRIMARY KEY (id_comentario),
    CONSTRAINT fk_comentario_usuario FOREIGN KEY (id_usuario) REFERENCES usuario(id_usuario),
    CONSTRAINT fk_comentario_publicacion FOREIGN KEY (id_publicacion) REFERENCES publicacion(id_publicacion)
);

CREATE TABLE servicio (
    id_servicio SERIAL NOT NULL,
    id_usuario INTEGER NOT NULL,
    nombre VARCHAR(50) NOT NULL,
    descripcion VARCHAR(200),
    fecha_servicio TIMESTAMP NOT NULL,
    lugar VARCHAR(40),
    CONSTRAINT pk_servicio PRIMARY KEY (id_servicio),
    CONSTRAINT fk_servicio_usuario FOREIGN KEY (id_usuario) REFERENCES usuario(id_usuario)
);

CREATE TABLE tipo_servicio (
    id_tipo_servicio SMALLSERIAL NOT NULL,
    nombre VARCHAR(30) NOT NULL,
    descripcion VARCHAR(200),
    CONSTRAINT pk_tipo_servicio PRIMARY KEY (id_tipo_servicio)
);

CREATE TABLE servicio_tipo_servicio (
    id_servicio INTEGER NOT NULL,
    id_tipo_servicio SMALLINT NOT NULL,
    CONSTRAINT pk_servicio_tipo_servicio PRIMARY KEY (id_servicio, id_tipo_servicio),
    CONSTRAINT fk_ser_tipo_servicio FOREIGN KEY (id_servicio) REFERENCES servicio(id_servicio),
    CONSTRAINT fk_ser_tipo_tiposer FOREIGN KEY (id_tipo_servicio) REFERENCES tipo_servicio(id_tipo_servicio)
);

CREATE TABLE tipo_producto (
    id_tipo_producto SMALLSERIAL NOT NULL,
    nombre VARCHAR(30) NOT NULL,
    descripcion VARCHAR(200),
    CONSTRAINT pk_tipo_producto PRIMARY KEY (id_tipo_producto)
);

CREATE TABLE producto (
    id_producto SERIAL NOT NULL,
    id_tipo_producto SMALLINT NOT NULL,
    id_usuario INTEGER NOT NULL,
    nombre VARCHAR(30) NOT NULL,
    descripcion VARCHAR(300),
    cantidad INTEGER NOT NULL,
    precio NUMERIC(10,2) NOT NULL,
    fecha_producto TIMESTAMP NOT NULL,
    CONSTRAINT pk_producto PRIMARY KEY (id_producto),
    CONSTRAINT fk_producto_tipo FOREIGN KEY (id_tipo_producto) REFERENCES tipo_producto(id_tipo_producto),
    CONSTRAINT fk_producto_usuario FOREIGN KEY (id_usuario) REFERENCES usuario(id_usuario),
    CONSTRAINT chk_producto_cantidad CHECK (cantidad >= 0 AND cantidad <= 1000),
    CONSTRAINT chk_producto_precio CHECK (precio >= 0 AND precio <= 20000000)
);

CREATE TABLE solicitud (
    id_solicitud SERIAL NOT NULL,
    id_usuario INTEGER NOT NULL,
    id_servicio INTEGER,
    id_producto INTEGER,
    cantidad INTEGER,
    fecha_solicitud TIMESTAMP NOT NULL,
    CONSTRAINT pk_solicitud PRIMARY KEY (id_solicitud),
    CONSTRAINT fk_solicitud_usuario FOREIGN KEY (id_usuario) REFERENCES usuario(id_usuario),
    CONSTRAINT fk_solicitud_servicio FOREIGN KEY (id_servicio) REFERENCES servicio(id_servicio),
    CONSTRAINT fk_solicitud_producto FOREIGN KEY (id_producto) REFERENCES producto(id_producto),
    CONSTRAINT chk_solicitud_cantidad CHECK (cantidad IS NULL OR (cantidad >= 0 AND cantidad <= 1000))
);

CREATE TABLE intercambio (
    id_intercambio SERIAL NOT NULL,
    id_prod_ofrecido INTEGER NOT NULL,
    id_prod_solicitado INTEGER NOT NULL,
    id_usuario INTEGER NOT NULL,
    fecha_intercambio TIMESTAMP NOT NULL,
    fecha_respuesta TIMESTAMP,
    cantidad INTEGER,
    respuesta CHAR(2),
    CONSTRAINT pk_intercambio PRIMARY KEY (id_intercambio),
    CONSTRAINT fk_inter_prod_ofrecido FOREIGN KEY (id_prod_ofrecido) REFERENCES producto(id_producto),
    CONSTRAINT fk_inter_prod_solic FOREIGN KEY (id_prod_solicitado) REFERENCES producto(id_producto),
    CONSTRAINT fk_inter_usuario FOREIGN KEY (id_usuario) REFERENCES usuario(id_usuario),
    CONSTRAINT chk_inter_prod_dif CHECK (id_prod_ofrecido <> id_prod_solicitado),
    CONSTRAINT chk_inter_cantidad CHECK (cantidad IS NULL OR (cantidad >= 0 AND cantidad <= 1000))
);

CREATE TABLE chat_privado (
    id_chat SERIAL NOT NULL,
    id_usuario_1 INTEGER NOT NULL,
    id_usuario_2 INTEGER NOT NULL,
    CONSTRAINT pk_chat_privado PRIMARY KEY (id_chat),
    CONSTRAINT fk_chat_usuario_1 FOREIGN KEY (id_usuario_1) REFERENCES usuario(id_usuario),
    CONSTRAINT fk_chat_usuario_2 FOREIGN KEY (id_usuario_2) REFERENCES usuario(id_usuario),
    CONSTRAINT chk_chat_usu_dif CHECK (id_usuario_1 <> id_usuario_2)
);

CREATE TABLE mensaje_privado (
    id_mensaje BIGSERIAL NOT NULL,
    id_chat INTEGER NOT NULL,
    id_usuario INTEGER NOT NULL,
    texto VARCHAR(2000),
    URL VARCHAR(2000),
    imagen CHARACTER VARYING(1000),
    audio CHARACTER VARYING(1000),
    fecha_mensaje TIMESTAMP NOT NULL,
    CONSTRAINT pk_mensaje_privado PRIMARY KEY (id_mensaje),
    CONSTRAINT fk_mensaje_chat FOREIGN KEY (id_chat) REFERENCES chat_privado(id_chat),
    CONSTRAINT fk_mensaje_usuario FOREIGN KEY (id_usuario) REFERENCES usuario(id_usuario)
);

CREATE TABLE tipo_evento (
    id_tipo_evento SMALLSERIAL NOT NULL,
    nombre VARCHAR(30) NOT NULL,
    descripcion VARCHAR(500),
    CONSTRAINT pk_tipo_evento PRIMARY KEY (id_tipo_evento)
);

CREATE TABLE evento (
    id_evento SERIAL NOT NULL,
    id_tipo_evento SMALLINT NOT NULL,
    id_usuario INTEGER NOT NULL,
    nombre VARCHAR(30) NOT NULL,
    contenido VARCHAR(600),
    fecha_evento TIMESTAMP NOT NULL,
    fecha_inicio TIMESTAMP NOT NULL,
    fecha_fin TIMESTAMP,
    lugar VARCHAR(30) NOT NULL,
    CONSTRAINT pk_evento PRIMARY KEY (id_evento),
    CONSTRAINT fk_evento_tipo FOREIGN KEY (id_tipo_evento) REFERENCES tipo_evento(id_tipo_evento),
    CONSTRAINT fk_evento_usuario FOREIGN KEY (id_usuario) REFERENCES usuario(id_usuario),
    CONSTRAINT chk_evento_fechas CHECK (fecha_fin IS NULL OR fecha_fin >= fecha_inicio)
);

CREATE TABLE usuario_evento (
    id_usuario INTEGER NOT NULL,
    id_evento INTEGER NOT NULL,
    fecha_ingreso TIMESTAMP NOT NULL,
    CONSTRAINT pk_usuario_evento PRIMARY KEY (id_usuario, id_evento),
    CONSTRAINT fk_usu_event_usuario FOREIGN KEY (id_usuario) REFERENCES usuario(id_usuario),
    CONSTRAINT fk_usu_event_evento FOREIGN KEY (id_evento) REFERENCES evento(id_evento)
);

CREATE TABLE tipo_notificacion (
    id_tipo_notificacion SMALLSERIAL NOT NULL,
    nombre VARCHAR(50) NOT NULL,
    id_publicacion BIGINT,
    id_comentario BIGINT,
    id_servicio INTEGER,
    id_producto INTEGER,
    id_solicitud INTEGER,
    id_intercambio INTEGER,
    id_evento INTEGER,
    id_mensaje BIGINT,
    CONSTRAINT pk_tipo_notificacion PRIMARY KEY (id_tipo_notificacion),
    CONSTRAINT fk_tiponot_publicacion FOREIGN KEY (id_publicacion) REFERENCES publicacion(id_publicacion),
    CONSTRAINT fk_tiponot_comentario FOREIGN KEY (id_comentario) REFERENCES comentario(id_comentario),
    CONSTRAINT fk_tiponot_servicio FOREIGN KEY (id_servicio) REFERENCES servicio(id_servicio),
    CONSTRAINT fk_tiponot_producto FOREIGN KEY (id_producto) REFERENCES producto(id_producto),
    CONSTRAINT fk_tiponot_solicitud FOREIGN KEY (id_solicitud) REFERENCES solicitud(id_solicitud),
    CONSTRAINT fk_tiponot_intercambio FOREIGN KEY (id_intercambio) REFERENCES intercambio(id_intercambio),
    CONSTRAINT fk_tiponot_evento FOREIGN KEY (id_evento) REFERENCES evento(id_evento),
    CONSTRAINT fk_tiponot_mensaje FOREIGN KEY (id_mensaje) REFERENCES mensaje_privado(id_mensaje)
);

CREATE TABLE notificacion (
    id_notificacion BIGSERIAL NOT NULL,
    id_tipo_notificacion SMALLINT NOT NULL,
    mensaje VARCHAR(50),
    fecha_notificacion TIMESTAMP NOT NULL,
    CONSTRAINT pk_notificacion PRIMARY KEY (id_notificacion),
    CONSTRAINT fk_notificacion_tipo FOREIGN KEY (id_tipo_notificacion) REFERENCES tipo_notificacion(id_tipo_notificacion)
);

CREATE TABLE notificacion_usuario (
    id_usuario INTEGER NOT NULL,
    id_notificacion BIGINT NOT NULL,
    CONSTRAINT pk_notificacion_usuario PRIMARY KEY (id_usuario, id_notificacion),
    CONSTRAINT fk_notif_usu_usuario FOREIGN KEY (id_usuario) REFERENCES usuario(id_usuario),
    CONSTRAINT fk_notif_usu_notificacion FOREIGN KEY (id_notificacion) REFERENCES notificacion(id_notificacion)
);
