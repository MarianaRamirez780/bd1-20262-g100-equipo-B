-- Creación de las tablas
-- 

CREATE TABLE rol (
	id_rol SMALLSERIAL PRIMARY KEY,
	nombre VARCHAR(30) NOT NULL,
	descripcion VARCHAR(200)
);

CREATE TABLE tipo_usuario (
	id_tipo_usuario SMALLSERIAL PRIMARY KEY,
	nombre VARCHAR(30) NOT NULL,
	descripcion VARCHAR(200)
);

CREATE TABLE dependencia (
	id_depa SMALLSERIAL PRIMARY KEY NOT NULL,
	nombre VARCHAR(50) NOT NULL
);

CREATE TABLE usuario (
	id_usuario SERIAL PRIMARY KEY NOT NULL,
	id_rol SMALLINT NOT NULL,
	nombre_1 VARCHAR(30) NOT NULL,
	nombre_2 VARCHAR(30),
	apellido_1 VARCHAR(30) NOT NULL,
	apellido_2 VARCHAR(30) NOT NULL,
	fecha_nac DATE NOT NULL,
	edad SMALLINT NOT NULL,
	sexo CHAR(1),
	correo VARCHAR(50) UNIQUE NOT NULL,
	contrasena VARCHAR(25) NOT NULL,
	telefono VARCHAR(15),
	fecha_registro TIMESTAMP NOT NULL,
	FOREIGN KEY (id_rol) REFERENCES rol(id_rol),
	CONSTRAINT chk_usuario_edad CHECK (edad >= 0 AND edad <= 150)
);

CREATE TABLE usuario_tipo_usuario (
	id_usuario INTEGER NOT NULL,
	id_tipo_usuario SMALLINT NOT NULL,
	PRIMARY KEY (id_usuario, id_tipo_usuario),
	FOREIGN KEY (id_usuario) REFERENCES usuario(id_usuario),
	FOREIGN KEY (id_tipo_usuario) REFERENCES tipo_usuario(id_tipo_usuario)
);

CREATE TABLE usuario_dependencia (
	id_depa SMALLINT NOT NULL,
	id_usuario INTEGER NOT NULL,
	PRIMARY KEY (id_depa, id_usuario),
	FOREIGN KEY (id_depa) REFERENCES dependencia(id_depa),
	FOREIGN KEY (id_usuario) REFERENCES usuario(id_usuario)
);

CREATE TABLE grupo (
	id_grupo SERIAL PRIMARY KEY NOT NULL,
	nombre VARCHAR(50) NOT NULL,
	id_perfil INTEGER NOT NULL,
	descipcion VARCHAR(500),
	fecha_creacion TIMESTAMP NOT NULL
);

CREATE TABLE perfil (
	id_perfil SERIAL PRIMARY KEY NOT NULL,
	id_usuario INTEGER NOT NULL,
	id_grupo INTEGER NOT NULL,
	biografia VARCHAR(500),
	foto BYTEA NOT NULL,
	perfil_extra JSONB,
	fecha_perfil TIMESTAMP NOT NULL,
	FOREIGN KEY (id_usuario) REFERENCES usuario(id_usuario),
	FOREIGN KEY (id_grupo) REFERENCES grupo(id_grupo)
);

ALTER TABLE grupo	-- Ya que antes no existía la tabla perfil
ADD CONSTRAINT fk_grupo_perfil FOREIGN KEY (id_perfil) REFERENCES perfil(id_perfil);

CREATE TABLE usuario_grupo (
	id_usuario INTEGER NOT NULL,
	id_grupo INTEGER NOT NULL,
	PRIMARY KEY (id_usuario, id_grupo),
	FOREIGN KEY (id_usuario) REFERENCES usuario(id_usuario),
	FOREIGN KEY (id_grupo) REFERENCES grupo(id_grupo)
);

CREATE TABLE publicacion (
	id_publicacion BIGSERIAL PRIMARY KEY NOT NULL,
	id_usuario INTEGER NOT NULL,
	id_perfil INTEGER NOT NULL,
	titulo VARCHAR(50) NOT NULL,
	texto VARCHAR(5000),
	URL VARCHAR(1000),
	imagen BYTEA,
	fecha_publicacion TIMESTAMP NOT NULL,
	FOREIGN KEY (id_usuario) REFERENCES usuario(id_usuario),
	FOREIGN KEY (id_perfil) REFERENCES perfil(id_perfil)
);

CREATE TABLE comentario (
	id_comentario BIGSERIAL PRIMARY KEY NOT NULL,
	id_usuario INTEGER NOT NULL,
	id_publicacion BIGINT NOT NULL,
	texto VARCHAR(2000),
	fecha_comentario TIMESTAMP NOT NULL,
	FOREIGN KEY (id_usuario) REFERENCES usuario(id_usuario),
	FOREIGN KEY (id_publicacion) REFERENCES publicacion(id_publicacion)
);

CREATE TABLE servicio (
	id_servicio SERIAL PRIMARY KEY NOT NULL,
	id_usuario INTEGER NOT NULL,
	nombre VARCHAR(50) NOT NULL,
	descripcion VARCHAR(200),
	fecha_servicio TIMESTAMP NOT NULL,
	lugar VARCHAR(40),
	FOREIGN KEY (id_usuario) REFERENCES usuario(id_usuario)
);

CREATE TABLE tipo_servicio (
	id_tipo_servicio SMALLSERIAL PRIMARY KEY NOT NULL,
	nombre VARCHAR(30) NOT NULL,
	descripcion VARCHAR(200)
);

CREATE TABLE servicio_tipo_servicio (
	id_servicio INTEGER NOT NULL,
	id_tipo_servicio SMALLINT NOT NULL,
	PRIMARY KEY (id_servicio, id_tipo_servicio),
	FOREIGN KEY (id_servicio) REFERENCES servicio(id_servicio),
	FOREIGN KEY (id_tipo_servicio) REFERENCES tipo_servicio(id_tipo_servicio)
);

CREATE TABLE tipo_producto (
	id_tipo_producto SMALLSERIAL PRIMARY KEY NOT NULL,
	nombre VARCHAR(30) NOT NULL,
	descripcion VARCHAR(200)
);

CREATE TABLE producto (
	id_producto SERIAL PRIMARY KEY NOT NULL,
	id_tipo_producto SMALLINT NOT NULL,
	id_usuario INTEGER NOT NULL,
	nombre VARCHAR(30) NOT NULL,
	descripcion VARCHAR(300),
	cantidad SMALLINT NOT NULL,
	precio NUMERIC(10,2) NOT NULL,
	fecha_producto TIMESTAMP NOT NULL,
	FOREIGN KEY (id_tipo_producto) REFERENCES tipo_producto(id_tipo_producto),
	FOREIGN KEY (id_usuario) REFERENCES usuario(id_usuario),
	CONSTRAINT chk_producto_cantidad CHECK (cantidad >= 0 AND cantidad <= 1000),
	CONSTRAINT chk_producto_precio CHECK (precio >= 0 AND precio <= 20000000)
);

CREATE TABLE solicitud (
	id_solicitud SERIAL NOT NULL,
	id_usuario INTEGER NOT NULL,
	id_servicio INTEGER,
	id_producto INTEGER,
	cantidad SMALLINT,
	fecha_solicitud TIMESTAMP NOT NULL,
	CONSTRAINT pk_solicitud PRIMARY KEY (id_solicitud),
	FOREIGN KEY (id_usuario) REFERENCES usuario(id_usuario),
	FOREIGN KEY (id_servicio) REFERENCES servicio(id_servicio),
	FOREIGN KEY (id_producto) REFERENCES producto(id_producto),
	CONSTRAINT chk_solicitud_cantidad CHECK (cantidad IS NULL OR (cantidad >= 0 AND cantidad <= 1000))
);

CREATE TABLE intercambio (
	id_intercambio SERIAL PRIMARY KEY NOT NULL,
	id_prod_ofrecido INTEGER NOT NULL,
	id_prod_solicitado INTEGER NOT NULL,
	id_usuario INTEGER NOT NULL,
	fecha_intercambio TIMESTAMP NOT NULL,
	fecha_respuesta TIMESTAMP,
	cantidad SMALLINT NOT NULL,
	respuesta BOOLEAN,
	FOREIGN KEY (id_prod_ofrecido) REFERENCES producto(id_producto),
	FOREIGN KEY (id_prod_solicitado) REFERENCES producto(id_producto),
	FOREIGN KEY (id_usuario) REFERENCES usuario(id_usuario),
	CONSTRAINT chk_inter_productos CHECK (id_prod_ofrecido <> id_prod_solicitado),
	CONSTRAINT chk_inter_cantidad CHECK (cantidad >= 0 AND cantidad <= 1000)
);

CREATE TABLE chat_privado (
	id_chat SERIAL PRIMARY KEY NOT NULL,
	id_usuario_1 INTEGER NOT NULL,
	id_usuario_2 INTEGER NOT NULL,
	FOREIGN KEY (id_usuario_1) REFERENCES usuario(id_usuario),
	FOREIGN KEY (id_usuario_2) REFERENCES usuario(id_usuario),
	CONSTRAINT chk_chat_usuarios CHECK (id_usuario_1 <> id_usuario_2)
);

CREATE TABLE mensaje_privado (
	id_mensaje BIGSERIAL PRIMARY KEY NOT NULL,
	id_chat INTEGER NOT NULL,
	id_usuario INTEGER NOT NULL,
	texto VARCHAR(2000),
	URL VARCHAR(2000),
	imagen BYTEA,
	audio BYTEA,
	fecha_mensaje TIMESTAMP NOT NULL,
	FOREIGN KEY (id_chat) REFERENCES chat_privado(id_chat),
	FOREIGN KEY (id_usuario) REFERENCES usuario(id_usuario)
);

CREATE TABLE tipo_evento (
	id_tipo_evento SMALLSERIAL PRIMARY KEY NOT NULL,
	nombre VARCHAR(30) NOT NULL,
	descripcion VARCHAR(500)
);

CREATE TABLE evento (
	id_evento SERIAL PRIMARY KEY NOT NULL,
	id_tipo_evento SMALLINT NOT NULL,
	id_usuario INTEGER NOT NULL,
	nombre VARCHAR(30) NOT NULL,
	contenido VARCHAR(600),
	fecha_evento TIMESTAMP NOT NULL,
	fecha_inicio TIMESTAMP NOT NULL,
	fecha_fin TIMESTAMP,
	lugar VARCHAR(30) NOT NULL,
	FOREIGN KEY (id_tipo_evento) REFERENCES tipo_evento(id_tipo_evento),
	FOREIGN KEY (id_usuario) REFERENCES usuario(id_usuario),
	CONSTRAINT chk_evento_fechas CHECK (fecha_fin IS NULL OR fecha_fin >= fecha_inicio)
);

CREATE TABLE usuario_evento (
	id_usuario INTEGER NOT NULL,
	id_evento INTEGER NOT NULL,
	fecha_ingreso TIMESTAMP NOT NULL,
	PRIMARY KEY (id_usuario, id_evento),
	FOREIGN KEY (id_usuario) REFERENCES usuario(id_usuario),
	FOREIGN KEY (id_evento) REFERENCES evento(id_evento)
);

CREATE TABLE tipo_notificacion (
	id_tipo_notificacion SMALLSERIAL PRIMARY KEY NOT NULL,
	nombre VARCHAR(50) NOT NULL,
	id_publicacion BIGINT,
	id_comentario BIGINT,
	id_servicio INTEGER,
	id_producto INTEGER,
	id_solicitud INTEGER,
	id_intercambio INTEGER,
	id_evento INTEGER,
	id_mensaje BIGINT,
	FOREIGN KEY (id_publicacion) REFERENCES publicacion(id_publicacion),
	FOREIGN KEY (id_comentario) REFERENCES comentario(id_comentario),
	FOREIGN KEY (id_servicio) REFERENCES servicio(id_servicio),
	FOREIGN KEY (id_producto) REFERENCES producto(id_producto),
	FOREIGN KEY (id_solicitud) REFERENCES solicitud(id_solicitud),
	FOREIGN KEY (id_intercambio) REFERENCES intercambio(id_intercambio),
	FOREIGN KEY (id_evento) REFERENCES evento(id_evento),
	FOREIGN KEY (id_mensaje) REFERENCES mensaje_privado(id_mensaje)
);

CREATE TABLE notificacion (
	id_notificacion BIGSERIAL PRIMARY KEY NOT NULL,
	id_tipo_notificacion SMALLINT NOT NULL,
	mensaje VARCHAR(50),
	fecha_notificacion TIMESTAMP NOT NULL,
	FOREIGN KEY (id_tipo_notificacion) REFERENCES tipo_notificacion(id_tipo_notificacion)
);

CREATE TABLE notificacion_usuario (
	id_usuario INTEGER NOT NULL,
	id_notificacion BIGINT NOT NULL,
	PRIMARY KEY (id_usuario, id_notificacion),
	FOREIGN KEY (id_usuario) REFERENCES usuario(id_usuario),
	FOREIGN KEY (id_notificacion) REFERENCES notificacion(id_notificacion)
);
